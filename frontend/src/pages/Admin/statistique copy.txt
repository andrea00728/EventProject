// Plugin personnalisé pour afficher le total au centre d'un graphique en beignet
const centerTextPlugin = {
  id: 'centerText',
  /**
   * @param {Chart<keyof ChartTypeRegistry, number[], unknown>} chart
   */
  beforeDraw(chart) {
    const { width, height, ctx } = chart;
    ctx.save();

    const fontSize = (height / 120).toFixed(2);
    ctx.font = `${fontSize}em sans-serif`;
    ctx.textBaseline = 'middle';
    ctx.fillStyle = '#cbd5e1'; // Couleur du texte, correspond au thème

    const total = chart.data.datasets[0].data.reduce((a, b) => {
      // S'assurer que 'a' et 'b' sont des nombres avant l'addition
      return (typeof a === 'number' ? a : 0) + (typeof b === 'number' ? b : 0);
    }, 0);
    const text = `Total: ${total}`;
    const textX = Math.round((width - ctx.measureText(text).width) / 2);
    const textY = height / 2;

    ctx.fillText(text, textX, textY);
    ctx.restore();
  }
};
